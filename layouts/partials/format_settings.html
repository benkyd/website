{{- /* . is a dict with focal, aperture, shutter, iso (any may be ""). Returns a consistently-formatted " · "-joined string, e.g. "50mm · f/5.6 · 1/1000s · ISO 100" - shared so manually-entered gear formats exactly like real EXIF does, regardless of how the value was typed in. */}} {{- $parts := slice }} {{- with .focal }}{{ $parts = $parts | append (printf "%smm" (strings.TrimSuffix "mm" .)) }}{{ end }} {{- with .aperture }}{{ $parts = $parts | append (printf "f/%s" (strings.TrimPrefix "f/" .)) }}{{ end }} {{- with .shutter }}{{ $parts = $parts | append (printf "%ss" (strings.TrimSuffix "s" .)) }}{{ end }} {{- with .iso }}{{ $parts = $parts | append (printf "ISO %s" (strings.TrimPrefix "ISO " .)) }}{{ end }} {{- return (delimit $parts " · ") }}
